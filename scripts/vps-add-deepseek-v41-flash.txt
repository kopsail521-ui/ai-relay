  cur.execute("SELECT value FROM options WHERE key=?",(k,)); r=cur.fetchone(); return r[0] if r else "{}"

mid="deepseek-v4.1-flash"

# force rewrite if missing/wrong
need=False
if abs(float(mr.get(mid) or 0)-0.225)>1e-9:
  mr[mid]=0.225; need=True
if abs(float(cr.get(mid) or 0)-4)>1e-9:
  cr[mid]=4; need=True
if need:
  for k,v in (("ModelRatio",mr),("CompletionRatio",cr)):
    cur.execute("SELECT key FROM options WHERE key=?",(k,))
    s=json.dumps(v,ensure_ascii=False,separators=(",",":"))
    if cur.fetchone() is None: cur.execute("INSERT INTO options(key,value) VALUES(?,?)",(k,s))
    else: cur.execute("UPDATE options SET value=? WHERE key=?",(s,k))
  con.commit()
  print("db_rewrote", True)
  import subprocess
  subprocess.call("sudo docker restart ai-relay-new-api 2>/dev/null || sudo docker restart new-api 2>/dev/null || true", shell=True)

else:
  print("db_rewrote", False)
con.close()
