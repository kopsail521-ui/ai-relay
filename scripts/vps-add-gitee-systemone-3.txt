# VPS：上架 3 个 Gitee System One 模型（SemIf-OpenJev-4B / DiffusionGemma-26B-A4B-it-Jev / laya-multilingual）
# 售价 = 上游成本 ×2：$0.027397 / $0.082192 / $0.00274（输入，输出 $0），标签「系统一模型」
# 前提：本机已 git push
set -e
cd /opt/ai-relay
sudo git fetch origin
sudo git checkout -- \
  config/gitee-selected-models.json \
  config/marketplace-model-copy.json \
  services/creem-moderation-proxy/marketplace-model-copy.json \
  services/gitee-passthrough \
  scripts/vps-add-gitee-systemone-3.py \
  || true
sudo git pull --ff-only origin main

# 1) New API：渠道 + 售价（ModelRatio/CompletionRatio）+ 广场单标签「系统一模型」
sudo docker run --rm \
  -v /opt/ai-relay/data/new-api:/data \
  -v /opt/ai-relay/scripts/vps-add-gitee-systemone-3.py:/tmp/add.py:ro \
  python:3.12-alpine python /tmp/add.py /data/one-api.db

# 2) Gitee 透传 rebuild（catalog +38 个模型 + 新 RULES 自愈）
sudo cp -a /opt/ai-relay/config/gitee-selected-models.json \
  /opt/ai-relay/services/gitee-passthrough/catalog.json
sudo docker build -t keyo-gitee-passthrough ./services/gitee-passthrough
sudo docker rm -f ai-relay-gitee-passthrough 2>/dev/null || true
sudo docker run -d --name ai-relay-gitee-passthrough --restart always --network host \
  --env-file /opt/ai-relay/.env.gitee \
  -v /opt/ai-relay/data/new-api:/data:rw \
  -e NEW_API_DB=/data/one-api.db \
  -e CATALOG=/app/catalog.json \
  keyo-gitee-passthrough
sleep 2
curl -sS http://127.0.0.1:3010/healthz; echo

# 3) 广场简介/标签 i18n：热更新 copy 后重启 moderation（不重建镜像）
MOD=$(sudo docker ps --format '{{.Names}}' | grep -E 'creem|moderation' | head -1 || true)
if [ -n "$MOD" ]; then
  sudo docker cp /opt/ai-relay/services/creem-moderation-proxy/marketplace-model-copy.json \
    "$MOD":/app/marketplace-model-copy.json
  sudo docker restart "$MOD"
  echo "moderation_copy_updated=$MOD"
else
  echo "WARN: moderation container not found — tag i18n may lag until next moderation deploy"
fi

echo "--- smoke ---"
# 2') 透传健康：模型数应为 38
curl -sS http://127.0.0.1:3010/healthz | grep -o '"models":[0-9]*'
# 3') 公开定价接口应出现 3 个新模型（ abilities + ratio 生效 ）
curl -sS https://www.keyoapi.xyz/api/pricing > /tmp/pricing.json
for m in SemIf-OpenJev-4B DiffusionGemma-26B-A4B-it-Jev laya-multilingual; do
  grep -q "$m" /tmp/pricing.json && echo "pricing_ok=$m" || echo "pricing_MISSING=$m"
done
# 4') /v1/systemone 路由应打到透传（未带 key 返回 401 而非 404）
curl -sS -o /dev/null -w "systemone_http=%{http_code}\n" -X POST https://www.keyoapi.xyz/v1/systemone \
  -H "Content-Type: application/json" -d '{}'
echo DONE_ADD_GITEE_SYSTEMONE3
