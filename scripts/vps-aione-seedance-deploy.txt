# Seedance aione SKU 上架：pull + AIONE key + rebuild + DB register + smoke
# 前置：在 VPS /opt/ai-relay/.env 写入 AIONE_API_KEY=sk-...（及可选 AIONE_BASE_URL）
set -e
cd /opt/ai-relay

sudo git checkout -- \
  config/apimart-selected-models.json \
  config/marketplace-model-copy.json \
  config/model-icon-map.json \
  services/apimart-passthrough/catalog.json \
  services/apimart-passthrough/server.mjs \
  services/creem-moderation-proxy/marketplace-model-copy.json \
  services/creem-moderation-proxy/model-icon-map.json \
  services/creem-moderation-proxy/server.mjs \
  || true

sudo git pull

# AIONE key → video container env（从根 .env 拷贝）
sudo touch /opt/ai-relay/.env.apimart
if ! sudo grep -q '^AIONE_API_KEY=.' /opt/ai-relay/.env.apimart 2>/dev/null; then
  if ! sudo grep -q '^AIONE_API_KEY=.' /opt/ai-relay/.env 2>/dev/null; then
    echo "ERROR: set AIONE_API_KEY in /opt/ai-relay/.env first" >&2
    exit 1
  fi
  sudo grep -E '^AIONE_(API_KEY|BASE_URL)=' /opt/ai-relay/.env | sudo tee -a /opt/ai-relay/.env.apimart >/dev/null
fi
if ! sudo grep -q '^AIONE_BASE_URL=' /opt/ai-relay/.env.apimart 2>/dev/null; then
  echo 'AIONE_BASE_URL=https://api.aione.help' | sudo tee -a /opt/ai-relay/.env.apimart >/dev/null
fi
echo "inventory_d_key=$(sudo grep -c '^AIONE_API_KEY=.' /opt/ai-relay/.env.apimart || true)"

sudo docker build -t keyo-apimart-passthrough ./services/apimart-passthrough
sudo docker rm -f ai-relay-apimart-passthrough 2>/dev/null || true
sudo docker run -d --name ai-relay-apimart-passthrough --restart always --network host \
  --env-file /opt/ai-relay/.env.apimart \
  -v /opt/ai-relay/data:/data:rw \
  -e NEW_API_DB=/data/new-api/one-api.db \
  -e NEW_API_BASE=http://127.0.0.1:3000 \
  -e CATALOG=/app/catalog.json \
  -e LISTEN_HOST=127.0.0.1 \
  -e PORT=3011 \
  keyo-apimart-passthrough
sleep 2
curl -sS http://127.0.0.1:3011/healthz; echo

sudo python3 /opt/ai-relay/scripts/vps-add-apimart-videos.py /opt/ai-relay/data/new-api/one-api.db

if sudo docker ps --format '{{.Names}}' | grep -q '^ai-relay-creem-moderation$'; then
  sudo docker cp /opt/ai-relay/services/creem-moderation-proxy/marketplace-model-copy.json \
    ai-relay-creem-moderation:/app/marketplace-model-copy.json 2>/dev/null || true
  sudo docker cp /opt/ai-relay/services/creem-moderation-proxy/model-icon-map.json \
    ai-relay-creem-moderation:/app/model-icon-map.json 2>/dev/null || true
  sudo docker cp /opt/ai-relay/services/creem-moderation-proxy/server.mjs \
    ai-relay-creem-moderation:/app/server.mjs 2>/dev/null || true
  sudo docker restart ai-relay-creem-moderation >/dev/null || true
fi
sudo docker restart ai-relay-new-api >/dev/null || true
sleep 3

KEY=$(python3 - <<'PY'
import sqlite3
k=sqlite3.connect("/opt/ai-relay/data/new-api/one-api.db").execute(
  "SELECT key FROM tokens WHERE id=2 AND status=1"
).fetchone()[0]
print(k if str(k).startswith("sk-") else "sk-"+k)
PY
)

echo "=== smoke seedance-2.0-720p-mini ==="
curl -sS -o /tmp/sd720.json -w "sd720=%{http_code}\n" -m 90 \
  -X POST https://www.keyoapi.xyz/v1/videos/generations \
  -H "Authorization: Bearer $KEY" -H "Content-Type: application/json" \
  -d '{"model":"seedance-2.0-720p-mini","prompt":"a red balloon floating in blue sky, cinematic","duration":5}'
python3 - <<'PY'
import re
t=open("/tmp/sd720.json","r",encoding="utf-8",errors="ignore").read()
t=re.sub(r"(?i)aione|aicopy|grsai|dakka|apimart|openlux|gitee|moark","provider",t)
print(t[:500])
PY
echo DONE_SEEDANCE_AIONE

# Always re-heal marketplace vendors/icons/免费 (video onboard must not leave stale meta)
bash /opt/ai-relay/scripts/vps-heal-marketplace-meta.txt || true
