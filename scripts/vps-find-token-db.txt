echo "=== C) 每个库里有没有这把 key（前缀 $PREF）==="
python3 - <<PY
import os, sqlite3, glob
bare="$BARE"
pref="$PREF"
paths=glob.glob("/opt/ai-relay/data/**/*.db", recursive=True)
# also common docker paths if present on host
for p in paths:
  try:
    sz=os.path.getsize(p)
    con=sqlite3.connect(f"file:{p}?mode=ro", uri=True)
    cur=con.cursor()
    tabs={r[0] for r in cur.execute("SELECT name FROM sqlite_master WHERE type='table'")}
    if "tokens" not in tabs:
      print(p, "size", sz, "NO_tokens_table")
      con.close(); continue
    n=cur.execute("SELECT COUNT(*) FROM tokens").fetchone()[0]
    exact=cur.execute("SELECT id,user_id,status,length(key) FROM tokens WHERE key=? OR key=?",(bare,"sk-"+bare)).fetchall()
    pref_hits=cur.execute("SELECT id,user_id,status,substr(key,1,16),length(key) FROM tokens WHERE key LIKE ? LIMIT 5",(pref+"%",)).fetchall()
    print(p, "size", sz, "tokens", n, "exact", exact, "prefix", pref_hits)
    con.close()
  except Exception as e:
    print(p, "ERR", e)
PY
