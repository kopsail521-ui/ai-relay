# 修复「接入文档」404（粘贴到阿里云 Workbench）
# 原因：首页/导航指向 /docs，线上 SPA(rc.27) 无该路由 → 客户端 404
# 真实文档：/brand/keyo-docs.html（已 200）

# 1) 首页按钮改指向真实文档
sudo sed -i 's|href="/docs"|href="/brand/keyo-docs.html"|g' /opt/ai-relay/static/brand/keyo-home.html
grep -n 'keyo-docs\|href="/docs"' /opt/ai-relay/static/brand/keyo-home.html | head

# 2) 导航「文档」等 /docs 入口统一跳转（保留现有 Caddy 结构）
sudo tee /etc/caddy/Caddyfile >/dev/null <<'EOF'
www.keyoapi.xyz {
	encode gzip

	handle_path /brand/* {
		root * /opt/ai-relay/static/brand
		file_server
	}

	# 线上无 /docs SPA 路由，统一到品牌文档页
	redir /docs /brand/keyo-docs.html permanent
	redir /docs/ /brand/keyo-docs.html permanent

	# 静态资源直连 New API，避免审核代理整包缓冲大 JS
	handle /static/* {
		reverse_proxy 127.0.0.1:3000 {
			header_up Accept-Encoding identity
		}
	}

	# Gitee 特殊能力（若透传未装，这些路径会 502，不影响官网页面）
	@gitee_special path /v1/images/object-detection* /v1/images/segmentation* /v1/images/pose-detection* /v1/images/upscaling* /v1/images/unwarping* /v1/images/mattings* /v1/async/* /v1/task/*
	handle @gitee_special {
		reverse_proxy 127.0.0.1:3010 {
			header_up Accept-Encoding identity
		}
	}

	# 其余走审核代理；若代理挂了可临时改成 3000
	handle {
		reverse_proxy 127.0.0.1:3001 {
			header_up Accept-Encoding identity
		}
	}
}
EOF

sudo caddy validate --config /etc/caddy/Caddyfile && sudo systemctl reload caddy

echo '---- verify ----'
curl -sS -m 5 -o /dev/null -w 'home_btn=%{http_code}\n' https://www.keyoapi.xyz/brand/keyo-home.html
curl -sS -m 5 -o /dev/null -w 'docs_html=%{http_code}\n' https://www.keyoapi.xyz/brand/keyo-docs.html
curl -sS -m 5 -o /dev/null -w 'docs_redir=%{http_code} loc=%{redirect_url}\n' https://www.keyoapi.xyz/docs
echo DONE_FIX_DOCS_404
