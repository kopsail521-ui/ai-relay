sudo python3 - <<'PY'
import sqlite3, time
db="/opt/ai-relay/data/new-api/one-api.db"
# Rerank: Gitee 返回「暂不支持该接口」
# gemma: 直打 ai.gitee.com 超时无响应
# veo: supply_unavailable
mids=[
  "veo_3_1-components",
  "Qwen3-VL-Reranker-2B",
  "Qwen3-VL-Reranker-8B",
  "gemma-4-26B-A4B-it",
]
con=sqlite3.connect(db)
cur=con.cursor()
cols={r[1] for r in cur.execute("PRAGMA table_info(models)")}
tabs={r[0] for r in cur.execute("SELECT name FROM sqlite_master WHERE type='table'")}
now=int(time.time())
for mid in mids:
  if "deleted_at" in cols:
    cur.execute("UPDATE models SET deleted_at=? WHERE model_name=? AND (deleted_at IS NULL OR deleted_at=0)", (now, mid))
    print(mid, "soft_delete", cur.rowcount)
  if "status" in cols:
    cur.execute("UPDATE models SET status=0 WHERE model_name=?", (mid,))
    print(mid, "status0", cur.rowcount)
  if "abilities" in tabs:
    cur.execute("DELETE FROM abilities WHERE model=?", (mid,))
    print(mid, "abilities_del", cur.rowcount)
  for cid, models in cur.execute("SELECT id, models FROM channels").fetchall():
    if not models or mid not in models: continue
    parts=[p.strip() for p in models.replace("\n",",").split(",") if p.strip() and p.strip()!=mid]
    cur.execute("UPDATE channels SET models=? WHERE id=?", (",".join(parts), cid))
    print(mid, "channel_strip", cid)
con.commit()
print("DONE_DELIST_DB")
con.close()
PY
