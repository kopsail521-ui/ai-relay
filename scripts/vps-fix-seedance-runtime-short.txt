echo "=== whitelist probe ALL catalog models ==="
python3 - <<'PY'
import json, subprocess, os
cat=json.load(open("/opt/ai-relay/services/apimart-passthrough/catalog.json",encoding="utf-8"))
key=open("/tmp/keyo-smoke-key.txt").read().strip()
fail=[]
for m in cat["models"]:
  mid=m["id"]
  body={"model":mid,"prompt":"a red balloon floating in blue sky","duration":5}
  # MiniMax needs its own fields; Seedance must NOT get landscape/480p
  if mid=="MiniMax-H3":
    body.update({"aspectRatio":"landscape","resolution":"480p"})
  elif "grok-imagine" in mid:
    body["image"]={"url":"https://www.keyoapi.xyz/brand/logo.svg"}
  p=subprocess.run([
    "curl","-sS","-m","60","-o","/tmp/vw.json","-w","%{http_code}",
    "-X","POST","https://www.keyoapi.xyz/v1/videos/generations",
    "-H",f"Authorization: Bearer {key}","-H","Content-Type: application/json",
    "-d",json.dumps(body,ensure_ascii=False)
  ],capture_output=True,text=True)
  code=p.stdout.strip() or "000"
  t=open("/tmp/vw.json",encoding="utf-8",errors="ignore").read()
  hard=("Unknown model" in t) or ("model_not_available" in t) or ("无权访问" in t) or ("无可用渠道" in t) or ("超分版" in t) or ("不卡脸" in t)
  # accept queued/submitted OR validation errors that are model-specific (e.g. grok needs image)
  soft_ok = (code in ("200","201")) or ("image" in t.lower() and "grok-imagine" in mid) or (code=="400" and "prompt" in t.lower() and "Unknown" not in t and "model_not_available" not in t)
  if hard or not (soft_ok or code in ("200","201","400","402","403","429","503")):
    # 400 without model_not_available can still be OK (param); only hard fails count
    if hard or code in ("000",):
      fail.append(f"{mid} http={code} {t[:140].replace(chr(10),' ')}")
      print("FAIL", mid, code)
    else:
      print("PASS", mid, code)
  else:
    print("PASS", mid, code)
print("WHITELIST", "OK" if not fail else "FAIL", "fail_n=", len(fail))
for x in fail: print(" ", x)
PY
