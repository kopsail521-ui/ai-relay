# 3) smoke：标签聚合里不再有 免费 / 视频按次；归类正确
curl -sS https://www.keyoapi.xyz/api/pricing -o /tmp/p_tags.json
python3 - <<'PY'
import json
from collections import Counter
j = json.load(open("/tmp/p_tags.json"))
rows = {r["model_name"]: r for r in j.get("data", [])}
c = Counter()
for r in rows.values():
    for p in str(r.get("tags") or "").split(","):
        if p.strip():
            c[p.strip()] += 1
assert "免费" not in c, "免费 tag still present"
assert "视频按次" not in c, "视频按次 tag still present"
grok = rows.get("grok-video-1.5") or rows.get("grok-1.5-video")
assert grok is not None, "grok video model missing"
assert "视频模型" in str(grok.get("tags") or ""), "grok not in 视频模型: %r" % grok.get("tags")
kf = rows.get("keyo-flash:free")
if kf is not None:
    assert "免费" not in str(kf.get("tags") or ""), "keyo-flash:free still tagged 免费"
print("tags:", dict(c.most_common()))
print("grok tags:", grok.get("tags"), "| keyo-flash:free tags:", (kf or {}).get("tags"))
print("SMOKE_PASS")
PY
rm -f /tmp/p_tags.json
