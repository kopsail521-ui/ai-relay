import base64, json, os, sqlite3, time, urllib.request
from pathlib import Path

BASE = "https://www.keyoapi.xyz"

def get_key():
    c = sqlite3.connect(f"file:{DB}?mode=ro", uri=True)
    rows = c.execute(
        "SELECT user_id, key, name FROM tokens WHERE status=1 AND deleted_at IS NULL "
        "ORDER BY CASE WHEN user_id=3 THEN 0 ELSE 1 END, id DESC LIMIT 20"
    ).fetchall()
    if not rows:
        raise SystemExit("NO_TOKEN")
    uid, key, name = rows[0]
    key = key if str(key).startswith("sk-") else "sk-" + key
    print("using_token", {"user_id": uid, "name": name, "key_prefix": key[:10] + "..."})
    return key

def fetch(url, dest):
    if dest.exists() and dest.stat().st_size > 1000:
        return dest
    print("download", url)
    urllib.request.urlretrieve(url, dest)
    return dest

def make_png(path, lines, font_path, size=36):
    from PIL import Image, ImageDraw, ImageFont
    font = ImageFont.truetype(str(font_path), size)
    pad, gap = 48, 28
    tmp = Image.new("RGB", (10, 10), "white")
    d0 = ImageDraw.Draw(tmp)
    widths, heights = [], []
    for line in lines:
        b = d0.textbbox((0, 0), line, font=font)
        widths.append(b[2] - b[0]); heights.append(b[3] - b[1])
    w = max(max(widths) + pad * 2, 720)
    h = max(pad * 2 + sum(heights) + gap * (len(lines) - 1), 400)
    img = Image.new("RGB", (w, h), "white")
    draw = ImageDraw.Draw(img)
    y = pad
    for i, line in enumerate(lines):
        draw.text((pad, y), line, fill=(20, 20, 20), font=font)
        y += heights[i] + gap
    img.save(path)
    print("wrote", path, img.size)

def pick_font(candidates):
    for p in candidates:
        if Path(p).exists():

    raise SystemExit("NO_FONT " + " | ".join(candidates))

def req(method, url, data=None, headers=None, form=None):
    h = {"Authorization": f"Bearer {KEY}"}
    if headers:
        h.update(headers)
    if form is not None:
        # multipart via curl for reliability
        import subprocess, tempfile
        cmd = ["curl", "-sS", "-X", method, url, "-H", f"Authorization: Bearer {KEY}"]
        for k, v in form.items():
            if k == "file":
                cmd += ["-F", f"file=@{v};type=image/png"]
            else:
                cmd += ["-F", f"{k}={v}"]
        out = subprocess.check_output(cmd, timeout=180)
        return 200, out
    body = None
    if data is not None:
        body = json.dumps(data).encode()
        h["Content-Type"] = "application/json"
    r = urllib.request.Request(url, data=body, headers=h, method=method)
    try:
        with urllib.request.urlopen(r, timeout=180) as resp:
            return resp.status, resp.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()

def flatten(obj):
    return json.dumps(obj, ensure_ascii=False)
