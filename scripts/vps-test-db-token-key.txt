set -e
cd /opt/ai-relay
KEY_SCRIPT='sk-EllutBxwL8EL62e300Vrv2yBXI9NhcdPMQ2csfdyjog76c9t'

echo "=== 1) 库里 id=2 ==="
python3 - <<'PY'
import sqlite3
con=sqlite3.connect("/opt/ai-relay/data/new-api/one-api.db")
row=con.execute("SELECT id,user_id,name,status,key FROM tokens WHERE id=2").fetchone()
if not row:
  print("NO_ID_2"); raise SystemExit(1)
tid,uid,name,status,k=row
script="EllutBxwL8EL62e300Vrv2yBXI9NhcdPMQ2csfdyjog76c9t"
print("id",tid,"user",uid,"name",name,"status",status,"len",len(k))
print("script_eq", k==script, "nocase_eq", k.lower()==script.lower())
mism=[i for i,(a,b) in enumerate(zip(k,script)) if a!=b]
print("mismatch_count", len(mism)+abs(len(k)-len(script)), "first_idxs", mism[:15])
print("DB_KEY=sk-"+k if not str(k).startswith("sk-") else "DB_KEY="+k)
con.close()
PY

echo "=== 3) mattings 用库里真 Key ==="

python3 - <<'PY'
import struct,zlib,pathlib
def chunk(t,d):
  return struct.pack('>I',len(d))+t+d+struct.pack('>I',zlib.crc32(t+d)&0xffffffff)
raw=b'\x00'+b'\x00\x00\x00'
png=b'\x89PNG\r\n\x1a\n'+chunk(b'IHDR',struct.pack('>IIBBBBB',1,1,8,2,0,0,0))+chunk(b'IDAT',zlib.compress(raw))+chunk(b'IEND',b'')
pathlib.Path('/opt/ai-relay/tmp/rmbg-test.png').write_bytes(png)
print('png_ok')
PY
curl -sS -o /opt/ai-relay/tmp/rmbg.out -w "http=%{http_code}\n" \

  -H "Authorization: Bearer $DB_KEY" \
  -F model=RMBG-2.0 \
  -F image=@/opt/ai-relay/tmp/rmbg-test.png
