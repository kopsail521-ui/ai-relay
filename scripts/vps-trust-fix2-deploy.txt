set -e
decode() {
  local name="$1" dest="$2" expect_sha="$3"
  tr -d '\n\r\t ' < "/tmp/vps-trust-${name}.b64" | base64 -d | gunzip | sudo tee "$dest" >/dev/null
  local got
  got=$(sha256sum "$dest" | awk '{print $1}')
  echo "${name}_bytes=$(wc -c < "$dest") sha=$got"
  test "$got" = "$expect_sha"
}
