# 写入 Media Tools 凭证并重建容器（密钥只进 VPS .env，勿提交 Git）
# 用法：把下面三行里的占位符换成真实值后整段粘贴，或直接用本机已生成的剪贴板口令
set -e
cd /opt/ai-relay
sudo git pull
git log -1 --oneline

# --- keys: replace PLACEHOLDER_* before paste if not using prefilled clipboard ---
AK='PLACEHOLDER_AK'
SK='PLACEHOLDER_SK'
sudo touch /opt/ai-relay/.env
sudo sed -i '/^VOLC_ACCESS_KEY=/d;/^VOLC_SECRET_KEY=/d;/^MEDIAKIT_API_KEY=/d' /opt/ai-relay/.env
printf 'VOLC_ACCESS_KEY=%s\nVOLC_SECRET_KEY=%s\n' "$AK" "$SK" | sudo tee -a /opt/ai-relay/.env >/dev/null
echo "volc_ak_set=$(sudo grep -c '^VOLC_ACCESS_KEY=.' /opt/ai-relay/.env || true)"
echo "volc_sk_set=$(sudo grep -c '^VOLC_SECRET_KEY=.' /opt/ai-relay/.env || true)"

sudo docker build -t ai-relay-volc-passthrough:local ./services/volc-passthrough
sudo docker rm -f ai-relay-volc-passthrough 2>/dev/null || true
sudo docker run -d --name ai-relay-volc-passthrough --restart unless-stopped --network host \
  --env-file /opt/ai-relay/.env \
  -e PORT=3012 -e LISTEN_HOST=0.0.0.0 \
  -e NEW_API_BASE=http://127.0.0.1:3000 \
  -e NEW_API_DB=/data/one-api.db \
  -v /opt/ai-relay/data/new-api:/data:rw \
  ai-relay-volc-passthrough:local
sleep 2
curl -sS -m 3 http://127.0.0.1:3012/healthz; echo
echo DONE_VOLC_ENV_KEYS
